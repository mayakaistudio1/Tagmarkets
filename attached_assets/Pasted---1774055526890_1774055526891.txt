// server/integrations/zoom-api.ts
export async function registerZoomParticipant(meetingId: string, user) {
  const payload = {
    email: user.email,
    first_name: user.firstName,
    last_name: user.lastName,
  };

  const res = await fetch(`https://api.zoom.us/v2/meetings/${meetingId}/registrants`, {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${process.env.ZOOM_JWT}`,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify(payload),
  });

  if (!res.ok) throw new Error(`Zoom registration failed: ${res.statusText}`);
  const data = await res.json();
  return data.join_url;
}
